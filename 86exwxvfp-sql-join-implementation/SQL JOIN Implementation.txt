-- INNER 
SELECT E.FNAME, E.LNAME, D.DNAME, P.PNAME
FROM Employee E
INNER JOIN Department D ON E.DNUM = D.DNUM
INNER JOIN Working_Hours W ON E.SSN = W.SSN
INNER JOIN Project P ON W.PNUM = P.PNUM;




-- LEFT 
SELECT E.FNAME, E.LNAME, D.DNAME, DEP.DEPNAME, DEP.DOB
FROM Employee E
LEFT JOIN Department D ON E.DNUM = D.DNUM
LEFT JOIN Dependent DEP ON E.SSN = DEP.SSN;




-- RIGHT
SELECT P.PNAME, D.DNAME, E.FNAME, E.LNAME
FROM Employee E
RIGHT JOIN Department D ON E.DNUM = D.DNUM
RIGHT JOIN Project P ON D.DNUM = P.DNUM;




-- SELF 
SELECT E.FNAME AS EMP_FNAME, E.LNAME AS EMP_LNAME,
       S.FNAME AS SUP_FNAME, S.LNAME AS SUP_LNAME,
       D.DNAME
FROM Employee E
JOIN Employee S ON E.SUPER_SSN = S.SSN
JOIN Department D ON E.DNUM = D.DNUM;
