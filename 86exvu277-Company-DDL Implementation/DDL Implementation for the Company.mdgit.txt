-- 1. Create the Employee table first
CREATE TABLE employees (
    SSN CHAR(9) PRIMARY KEY,
    Fname VARCHAR(50),
    Lname VARCHAR(50),
    Birth_Date DATE,
    Gender CHAR(1),
    Supervisor_SSN CHAR(9),
    DNUM INT
);

-- 2. Create the Department table
CREATE TABLE departments (
    DNUM INT PRIMARY KEY,
    DName VARCHAR(100),
    Mgr_SSN CHAR(9),
    Hiring_Date DATE,
    FOREIGN KEY (Mgr_SSN) REFERENCES employees(SSN)
);

-- 3. Create the Department Locations table
CREATE TABLE dept_locations (
    DNUM INT,
    Location VARCHAR(100),
    PRIMARY KEY (DNUM, Location),
    FOREIGN KEY (DNUM) REFERENCES departments(DNUM)
);

-- 4. Create the Project table
CREATE TABLE projects (
    PNumber INT PRIMARY KEY,
    Pname VARCHAR(100),
    Location VARCHAR(100),
    City VARCHAR(100),
    DNUM INT,
    FOREIGN KEY (DNUM) REFERENCES departments(DNUM)
);

-- 5. Create the Works_On table 
CREATE TABLE works_on (
    ESSN CHAR(9),
    PNumber INT,
    Working_Hours DECIMAL(5,2),
    PRIMARY KEY (ESSN, PNumber),
    FOREIGN KEY (ESSN) REFERENCES employees(SSN),
    FOREIGN KEY (PNumber) REFERENCES projects(PNumber)
);

--6. Create the Dependent table 
CREATE TABLE dependent (
    ESSN CHAR(9),
    Dependent_Name VARCHAR(100),
    Gender CHAR(1),
    Birth_Date DATE,
    PRIMARY KEY (ESSN, Dependent_Name),
    FOREIGN KEY (ESSN) REFERENCES employees(SSN) ON DELETE CASCADE
);

-- 7. Add the remaining links to the Employee table
ALTER TABLE employees ADD FOREIGN KEY (DNUM) REFERENCES departments(DNUM);
ALTER TABLE employees ADD FOREIGN KEY (Supervisor_SSN) REFERENCES employees(SSN);
